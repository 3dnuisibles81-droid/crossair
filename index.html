import tkinter as tk import sys class CrosshairApp: def _init_(self, root): self.root = root self.root.title("Crossair Pro") # Configuration de la fenêtre : Transparente, sans bordure, toujours au premier plan self.root.attributes('-topmost', True) self.root.attributes('-transparentcolor', 'black') self.root.config(bg='black') self.root.overrideredirect(True) # Enlève les bordures de la fenêtre # Mettre la fenêtre en plein écran self.root.geometry(f"{self.root.winfo_screenwidth()}x{self.root.winfo_screenheight()}+0+0") # Variables du viseur self.color = "#00FF00" # Vert par défaut self.size = 10 self.thickness = 2 self.gap = 5 self.is_active = True # Création du canvas pour dessiner self.canvas = tk.Canvas(self.root, bg='black', highlightthickness=0) self.canvas.pack(fill=tk.BOTH, expand=True) # Calcul du centre de l'écran self.screen_width = self.root.winfo_screenwidth() self.screen_height = self.root.winfo_screenheight() self.center_x = self.screen_width // 2 self.center_y = self.screen_height // 2 # Raccourcis clavier self.root.bind('', self.quit_app) self.root.bind('', self.toggle_crosshair) # Barre espace pour cacher/montrer # Dessiner le viseur initial self.draw_crosshair() # Boucle pour maintenir la fenêtre au premier plan self.keep_on_top() def draw_crosshair(self): self.canvas.delete("all") if not self.is_active: return # Dessin des 4 traits # Haut self.canvas.create_line( self.center_x, self.center_y - self.gap, self.center_x, self.center_y - self.gap - self.size, fill=self.color, width=self.thickness ) # Bas self.canvas.create_line( self.center_x, self.center_y + self.gap, self.center_x, self.center_y + self.gap + self.size, fill=self.color, width=self.thickness ) # Gauche self.canvas.create_line( self.center_x - self.gap, self.center_y, self.center_x - self.gap - self.size, self.center_y, fill=self.color, width=self.thickness ) # Droite self.canvas.create_line( self.center_x + self.gap, self.center_y, self.center_x + self.gap + self.size, self.center_y, fill=self.color, width=self.thickness ) # Point central optionnel (décommente si tu en veux un) # self.canvas.create_oval( # self.center_x - 1, self.center_y - 1, # self.center_x + 1, self.center_y + 1, # fill=self.color, outline=self.color # ) def toggle_crosshair(self, event): self.is_active = not self.is_active self.draw_crosshair() def keep_on_top(self): self.root.attributes('-topmost', True) self.root.after(1000, self.keep_on_top) def quit_app(self, event): self.root.destroy() sys.exit() if _name_ == "_main_": root = tk.Tk() app = CrosshairApp(root) root.mainloop()
